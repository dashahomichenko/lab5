# 1. GET картинки, якої немає в кеші (200 OK, X-Cache: MISS)
curl -i http://127.0.0.1:3000/200 -o /dev/null

# 2. Повторний GET тієї самої картинки з кешу (200 OK, X-Cache: HIT)
curl -I http://127.0.0.1:3000/200

# 3. GET неіснуючого коду стану (404 Not Found)
curl -i http://127.0.0.1:3000/999

# 4. PUT власної картинки в кеш (201 Created)
curl -i -X PUT --data-binary @./cache/200.jpg http://127.0.0.1:3000/201

# 5. GET картинки після PUT (200 OK, X-Cache: HIT)
curl -I http://127.0.0.1:3000/201

# 6. DELETE існуючої картинки з кешу (200 OK)
curl -i -X DELETE http://127.0.0.1:3000/201

# 7. Повторний DELETE картинки, якої вже немає в кеші (404 Not Found)
curl -i -X DELETE http://127.0.0.1:3000/201

# 8. Запит із недозволеним методом POST (405 Method Not Allowed)
curl -i -X POST http://127.0.0.1:3000/200
